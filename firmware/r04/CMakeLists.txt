cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
  if(DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
  else()
    message(FATAL_ERROR "Set PICO_SDK_PATH to the Raspberry Pi Pico SDK checkout")
  endif()
endif()
set(PICO_BOARD pico CACHE STRING "RP2040 board")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(c07_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
set(C07_CORE_DIR "${CMAKE_CURRENT_LIST_DIR}/src/core" CACHE PATH "Portable c07 core")
add_executable(c07_firmware src/main.c src/board.c src/startup.c src/pwm_engine.c src/commutation.c ${C07_CORE_DIR}/c07_control.c)
target_include_directories(c07_firmware PRIVATE ${C07_CORE_DIR} ${CMAKE_CURRENT_LIST_DIR}/src)
target_compile_options(c07_firmware PRIVATE -Wall -Wextra -Werror)
pico_generate_pio_header(c07_firmware ${CMAKE_CURRENT_LIST_DIR}/src/c07_pwm.pio)
target_link_libraries(c07_firmware pico_stdlib hardware_spi hardware_pio hardware_dma hardware_clocks hardware_watchdog hardware_timer hardware_irq m)
pico_enable_stdio_usb(c07_firmware 1)
pico_enable_stdio_uart(c07_firmware 0)
target_compile_definitions(c07_firmware PRIVATE PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 C07_REAL_TYPE=double C07_SELECTION_DIAGNOSTICS=0 PICO_STACK_SIZE=0x1000)
set(C07_TEST_BACKGROUND_START_OFFSET_US 0 CACHE STRING "Diagnostic only: delay background work within a timer period")
target_compile_definitions(c07_firmware PRIVATE C07_TEST_BACKGROUND_START_OFFSET_US=${C07_TEST_BACKGROUND_START_OFFSET_US})
set(C07_SELECTION_METHOD 2 CACHE STRING "Evaluation method only")
set(C07_SELECTION_STEP_MODE 2 CACHE STRING "Evaluation step rule")
set(C07_SELECTION_PERIOD_US 20000 CACHE STRING "Evaluation MPPT minimum period")
set(C07_SELECTION_SETTLE_US 5000 CACHE STRING "Evaluation settle confirmation")
set(C07_SELECTION_WINDOW_US 5000 CACHE STRING "Evaluation averaging window")
set(C07_SELECTION_MIN_V .05 CACHE STRING "Evaluation minimum step")
set(C07_SELECTION_MAX_V .5 CACHE STRING "Evaluation maximum step")
set(C07_SELECTION_GAIN_V .5 CACHE STRING "Evaluation step gain")
set(C07_SELECTION_POWER_DB_W .2 CACHE STRING "Evaluation power deadband")
set(C07_SELECTION_VOLTAGE_DB_V .02835360440340909 CACHE STRING "Evaluation voltage deadband")
set(C07_SELECTION_CURRENT_DB_A .01220703125 CACHE STRING "Evaluation current deadband")
set(C07_SELECTION_SLEW_A_PER_S 0 CACHE STRING "Optional current target slew; 0 disables normal-operation slew")
foreach(setting METHOD STEP_MODE PERIOD_US SETTLE_US WINDOW_US MIN_V MAX_V GAIN_V POWER_DB_W VOLTAGE_DB_V CURRENT_DB_A SLEW_A_PER_S)
  target_compile_definitions(c07_firmware PRIVATE C07_SELECTION_${setting}=${C07_SELECTION_${setting}})
endforeach()
pico_add_extra_outputs(c07_firmware)
